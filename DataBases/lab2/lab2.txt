-- задание 1
-- Сделать запрос для получения атрибутов из указанных таблиц, применив фильтры по указанным условиям:
-- Таблицы: Н_ЛЮДИ, Н_СЕССИЯ.
-- Вывести атрибуты: Н_ЛЮДИ.ОТЧЕСТВО, Н_СЕССИЯ.ЧЛВК_ИД.
-- Фильтры (AND):
-- a) Н_ЛЮДИ.ИМЯ < Александр.
-- b) Н_СЕССИЯ.УЧГОД < 2001/2002.
-- c) Н_СЕССИЯ.УЧГОД = 2001/2002.
-- Вид соединения: INNER JOIN.
select "Н_ЛЮДИ"."ОТЧЕСТВО", "Н_СЕССИЯ"."ЧЛВК_ИД" from "Н_ЛЮДИ"
    INNER JOIN "Н_СЕССИЯ" on "Н_ЛЮДИ"."ИД" = "Н_СЕССИЯ"."ЧЛВК_ИД"
    where ("Н_ЛЮДИ"."ИМЯ" < 'Александр' AND "Н_СЕССИЯ"."УЧГОД" < '2001/2002' AND "Н_СЕССИЯ"."УЧГОД" = '2001/2002');


-- задание 2
-- Сделать запрос для получения атрибутов из указанных таблиц, применив фильтры по указанным условиям:
-- Таблицы: Н_ЛЮДИ, Н_ВЕДОМОСТИ, Н_СЕССИЯ.
-- Вывести атрибуты: Н_ЛЮДИ.ОТЧЕСТВО, Н_ВЕДОМОСТИ.ИД, Н_СЕССИЯ.ДАТА.
-- Фильтры (AND):
-- a) Н_ЛЮДИ.ИМЯ > Ярослав.
-- b) Н_ВЕДОМОСТИ.ИД < 1250981.
-- Вид соединения: RIGHT JOIN.
SELECT "Н_ЛЮДИ"."ОТЧЕСТВО", "Н_ВЕДОМОСТИ"."ИД", "Н_СЕССИЯ"."ДАТА" FROM "Н_ЛЮДИ"
    RIGHT JOIN "Н_ВЕДОМОСТИ" ON "Н_ЛЮДИ"."ИД" = "Н_ВЕДОМОСТИ"."ЧЛВК_ИД"
    RIGHT JOIN "Н_СЕССИЯ" ON "Н_ВЕДОМОСТИ"."ЧЛВК_ИД" = "Н_СЕССИЯ"."ЧЛВК_ИД"
WHERE "Н_ЛЮДИ"."ИМЯ" > 'Ярослав' AND "Н_ВЕДОМОСТИ"."ИД" < 1250981;


-- задание 3
-- Вывести число студентов ФКТИУ, которые старше 25 лет.
select count(*) AS "Количество студентов ФКТиУ, старше 25 лет"
FROM "Н_ЛЮДИ"
         INNER JOIN "Н_УЧЕНИКИ" ON "Н_ЛЮДИ"."ИД" = "Н_УЧЕНИКИ"."ЧЛВК_ИД"
         INNER JOIN "Н_ПЛАНЫ" ON "Н_УЧЕНИКИ"."ПЛАН_ИД" = "Н_ПЛАНЫ"."ИД"
         INNER JOIN "Н_ОТДЕЛЫ" ON "Н_ПЛАНЫ"."ОТД_ИД" = "Н_ОТДЕЛЫ"."ИД"
where ((extract(YEAR FROM "КОНЕЦ") - extract(YEAR FROM "ДАТА_РОЖДЕНИЯ") - "КУРС") > 25 OR
       ((extract(YEAR FROM "НАЧАЛО") - extract(YEAR FROM "ДАТА_РОЖДЕНИЯ") - "КУРС") + 6 > 25 AND
        (extract(YEAR FROM "КОНЕЦ") - extract(YEAR FROM "ДАТА_РОЖДЕНИЯ") - "КУРС") < 25)) AND
    "Н_ОТДЕЛЫ"."КОРОТКОЕ_ИМЯ" = 'КТиУ';


-- задание 4
-- Выдать различные фамилии преподавателей и число людей с каждой из этих фамилий, ограничив список фамилиями, встречающимися ровно 10 раз на на очной форме обучения.
select "Н_ЛЮДИ"."ФАМИЛИЯ", count("Н_ЛЮДИ"."ФАМИЛИЯ") from "Н_ЛЮДИ" where "Н_ЛЮДИ"."ФАМИЛИЯ" in
    (select "Н_ЛЮДИ"."ФАМИЛИЯ" from "Н_ЛЮДИ"
        left join "Н_УЧЕНИКИ" on "Н_ЛЮДИ"."ИД" = "Н_УЧЕНИКИ"."ЧЛВК_ИД"
            join (select "Н_ЛЮДИ"."ФАМИЛИЯ", count("Н_ЛЮДИ"."ФАМИЛИЯ") from "Н_ЛЮДИ"
                join (select * from "Н_УЧЕНИКИ" join "Н_ПЛАНЫ" on "Н_УЧЕНИКИ"."ПЛАН_ИД" = "Н_ПЛАНЫ"."ИД"
                    join "Н_ФОРМЫ_ОБУЧЕНИЯ" on "Н_ПЛАНЫ"."ФО_ИД" = "Н_ФОРМЫ_ОБУЧЕНИЯ"."ИД"
                    WHERE "Н_ФОРМЫ_ОБУЧЕНИЯ"."НАИМЕНОВАНИЕ" = 'Очная') as "ОЧНИКИ"
                on "Н_ЛЮДИ"."ИД" = "ОЧНИКИ"."ЧЛВК_ИД"
            GROUP BY "Н_ЛЮДИ"."ФАМИЛИЯ"
            having count("Н_ЛЮДИ"."ФАМИЛИЯ") = 10) as "ляляля"
        on "Н_ЛЮДИ"."ФАМИЛИЯ" = "ляляля"."ФАМИЛИЯ"
    WHERE "Н_УЧЕНИКИ"."ЧЛВК_ИД" is null)
GROUP BY "Н_ЛЮДИ"."ФАМИЛИЯ";


-- задание 5
-- Выведите таблицу со средними оценками студентов группы 4100 (Номер, ФИО, Ср_оценка), у которых средняя оценка меньше максимальной оценк(е|и) в группе 1101.
SELECT "Н_ЛЮДИ"."ИД", "ФАМИЛИЯ", "ИМЯ", "ОТЧЕСТВО", AVG(CAST("ОЦЕНКА" AS NUMERIC)) AS "СРЕДНЯЯ_ОЦЕНКА"
FROM "Н_ЛЮДИ"
JOIN "Н_УЧЕНИКИ" ON "Н_УЧЕНИКИ"."ЧЛВК_ИД" = "Н_ЛЮДИ"."ИД" AND "Н_УЧЕНИКИ"."ГРУППА" = '4100'
JOIN "Н_ВЕДОМОСТИ" ON "Н_ВЕДОМОСТИ"."ЧЛВК_ИД" = "Н_ЛЮДИ"."ИД" AND "ОЦЕНКА" NOT IN ('осв', 'неявка', 'зачет', 'незач', '99')
GROUP BY "Н_ЛЮДИ"."ИД", "ФАМИЛИЯ", "ИМЯ", "ОТЧЕСТВО"
HAVING AVG(CAST ("ОЦЕНКА" AS NUMERIC)) < (SELECT MAX(CAST ("ОЦЕНКА" AS NUMERIC)) AS MARK
    FROM "Н_УЧЕНИКИ"
    JOIN "Н_ВЕДОМОСТИ" ON "Н_УЧЕНИКИ"."ГРУППА" = '1101'
    AND "Н_УЧЕНИКИ"."ЧЛВК_ИД" = "Н_ВЕДОМОСТИ"."ЧЛВК_ИД"
    AND "ОЦЕНКА" NOT IN('осв', 'неявка', 'зачет', 'незач', '99')
);


-- задание 6
-- Получить список студентов, зачисленных ровно первого сентября 2012 года на первый курс очной формы обучения. В результат включить:
-- номер группы;
-- номер, фамилию, имя и отчество студента;
-- номер и состояние пункта приказа;
-- Для реализации использовать подзапрос с EXISTS.
select "Н_УЧЕНИКИ"."ГРУППА", "Н_ЛЮДИ"."ФАМИЛИЯ", "Н_ЛЮДИ"."ИМЯ", "Н_ЛЮДИ"."ОТЧЕСТВО", "Н_УЧЕНИКИ"."П_ПРКОК_ИД" from "Н_УЧЕНИКИ"
    join "Н_ПЛАНЫ" on "Н_УЧЕНИКИ"."ПЛАН_ИД" = "Н_ПЛАНЫ"."ИД"
    join "Н_ЛЮДИ" on "Н_УЧЕНИКИ"."ЧЛВК_ИД" = "Н_ЛЮДИ"."ИД"
where extract(YEAR FROM "НАЧАЛО") = 2012 AND "Н_ПЛАНЫ"."КУРС" = 1
AND EXISTS( SELECT * FROM "Н_ФОРМЫ_ОБУЧЕНИЯ" WHERE "Н_ФОРМЫ_ОБУЧЕНИЯ"."ИД" = "Н_УЧЕНИКИ"."ВИД_ОБУЧ_ИД" AND "Н_ФОРМЫ_ОБУЧЕНИЯ"."НАИМЕНОВАНИЕ" = 'Очная');



-- задание 7
-- Сформировать запрос для получения числа в группе No 3100 троечников.
select count(*) AS "Троечники 3100" from
    (select "Н_УЧЕНИКИ"."ИД" from "Н_УЧЕНИКИ"
        join "Н_ВЕДОМОСТИ" on "Н_ВЕДОМОСТИ"."ЧЛВК_ИД" = "Н_УЧЕНИКИ"."ЧЛВК_ИД" and "Н_ВЕДОМОСТИ"."ОЦЕНКА" not in ('осв', 'неявка', 'зачет', 'незач', '99')
    where "Н_УЧЕНИКИ"."ГРУППА" = '3100'
    group by "Н_УЧЕНИКИ"."ИД"
    HAVING (3.5 > avg(cast("Н_ВЕДОМОСТИ"."ОЦЕНКА" as INT))) and 2.5 < avg(cast("Н_ВЕДОМОСТИ"."ОЦЕНКА" as INT))) as avg_mark;
